RLS bloqueando leitura de usuários
A política RLS do Supabase está impedindo que admins vejam todos os usuários. Execute este SQL no Supabase → SQL Editor:
-- Permite admins lerem todos os usuários
CREATE POLICY "admins_read_all_users"
ON public.users FOR SELECT
USING (
  auth.uid() IN (
    SELECT id FROM public.users WHERE is_admin = true
  )
  OR auth.uid() = id
);

-- Permite qualquer autenticado inserir/atualizar o próprio perfil
CREATE POLICY "users_upsert_own"
ON public.users FOR INSERT
WITH CHECK (auth.uid() = id);

CREATE POLICY "users_update_own"
ON public.users FOR UPDATE
USING (auth.uid() = id);

-- Permite admins atualizarem qualquer usuário
CREATE POLICY "admins_update_all"
ON public.users FOR UPDATE
USING (
  auth.uid() IN (
    SELECT id FROM public.users WHERE is_admin = true
  )
);
